# deepstream-app config: RF-DETR Nano on the 720p sample video.
# Writes an annotated video and per-frame detections (KITTI text files) for the parity test.
# Run inside the DeepStream container with the repo mounted at /work:
#     deepstream-app -c /work/configs/deepstream_app_nano.txt

[application]
enable-perf-measurement=1
perf-measurement-interval-sec=5
# One text file per frame: "<label> 0.0 0 0.0 left top right bottom ... confidence", in video pixels
gie-kitti-output-dir=/work/data/ds_out/kitti

[source0]
enable=1
# 3 = URI source (decodebin under the hood)
type=3
uri=file:///opt/nvidia/deepstream/deepstream/samples/streams/sample_720p.mp4
num-sources=1
gpu-id=0

[streammux]
gpu-id=0
batch-size=1
batched-push-timeout=40000
# Same size as the video, so the muxer doesn't rescale frames before nvinfer does
width=1280
height=720
live-source=0

[primary-gie]
enable=1
gpu-id=0
batch-size=1
gie-unique-id=1
config-file=/work/models/nano/config_infer_rfdetr.txt

[osd]
enable=1
gpu-id=0
border-width=2
text-size=12
text-color=1;1;1;1
text-bg-color=0.3;0.3;0.3;1
font=Serif

[tiled-display]
enable=0

[sink0]
enable=1
# 3 = encode to file
type=3
# 1 = mp4, 1 = H.264, 0 = hardware encoder
container=1
codec=1
enc-type=0
sync=0
bitrate=4000000
output-file=/work/data/ds_out/rfdetr_nano.mp4

[tests]
file-loop=0
